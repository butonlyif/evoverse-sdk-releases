Efinity 设计规范

Efinity 设计规范

基于 Efinity 2026.1 工具链和 EvoVerse 项目实际 RTL 代码总结。 数据来源：efx_map --help、efx_run.py、项目快照 RTL 代码、Golden Top 参考设计。

1. RTL 编码规范

1.1 文件结构

`default_nettype none
`timescale 1ns / 1ps

//============================================================================
// 模块名称: <module_name>
// 功能描述: <中文描述>
// 接口说明: <端口说明>
//============================================================================
module module_name (
    input  wire clk_i,
    input  wire rst_n_i,
    output wire sig_o
);

    // ---- 内部信号 ----
    reg [WIDTH-1:0] reg_name_r;

    // ---- 组合逻辑 / 时序逻辑 ----
    // ...

endmodule

`default_nettype wire

要点： - 文件头必须是 `default_nettype none，结尾恢复 `default_nettype wire - 禁止隐式 net 类型，所有 wire 必须显式声明 - 时间精度统一 1ns / 1ps

1.2 命名规范

类型 后缀 示例
端口输入 _i clk_i, rst_n_i, enable_i
端口输出 _o count_o, pwm_o, led_o
内部寄存器 _r count_r, state_r
低有效信号 _n rst_n, btn_n
时钟信号 clk_ clk_50m, sys_clk
参数常量 UPPER_CASE WIDTH, DEPTH

1.3 端口声明

所有端口必须显式指定方向 AND 类型：

// 正确
module top (
    input  wire       clk50_in,
    input  wire       rst_n_i,
    output wire       led_user,
    output reg  [7:0] count_o
);

// 错误 — 隐式 net 类型
module top (input clk50_in, output led_user);

1.4 参数化

module counter #(
    parameter WIDTH = 8
) (
    input  wire              clk_i,
    input  wire              rst_n_i,
    output wire [WIDTH-1:0]  count_o
);
    reg [WIDTH-1:0] count_r;
    // ...
endmodule

1.5 例化

// 正确 — 显式命名端口连接
counter_top #(
    .WIDTH(16)
) u_counter (
    .clk_i   (sys_clk),
    .rst_n_i (rst_n),
    .count_o (led_count)
);

// 禁止 — .* 隐式连接

1.6 分区注释

// ---- 异步复位同步释放 ----
reg [2:0] rst_sync;
always @(posedge sys_clk or negedge rst_n_i) begin
    // ...
end

// ---- 26 位计数器 ----
reg [25:0] blink_cnt;
always @(posedge sys_clk or negedge rst_n) begin
    // ...
end

// ---- LED 驱动 ----
assign led_user = blink_cnt[25];

2. 复位设计

2.1 异步复位同步释放（强制模式）

// 3 级同步器：异步断言，同步释放
reg [2:0] rst_sync;
always @(posedge sys_clk or negedge rst_n_i) begin
    if (!rst_n_i)
        rst_sync <= 3'b000;
    else
        rst_sync <= {rst_sync[1:0], 1'b1};
end
wire rst_n = rst_sync[2];

2.2 与 PLL Lock 合并

// PLL 未锁定时保持系统复位
wire async_rst_n = rst_n_i & pll_locked;

2.3 子模块复位

always @(posedge clk_i or negedge rst_n_i) begin
    if (!rst_n_i)
        count_r <= '0;
    else
        count_r <= count_r + 1'b1;
end

2.4 复位覆盖原则

  • 所有时序逻辑（寄存器）必须有复位
  • 组合逻辑的 assign 不直接复位，由驱动的寄存器间接复位
  • 复位信号极性统一为低有效（_n 后缀）

3. CDC 跨时钟域处理

3.1 单 bit 控制信号同步

// 2-3 级移位寄存器同步器
reg [2:0] sync;
always @(posedge dst_clk) begin
    sync <= {sync[1:0], src_signal};
end
wire dst_signal = sync[2];

3.2 多 bit 数据总线

  • 使用异步 FIFO（Gray 码指针）跨时钟域
  • 禁止直接对多 bit 总线使用移位寄存器同步

3.3 SDC 中的 CDC 声明

# 所有异步时钟域之间声明
set_clock_groups -asynchronous \
    -group {clk_50m} \
    -group {clk_33_ddr} \
    -group {clk_27_mipi_a clk_27_mipi_b}

# CDC 同步链首级 false path
set_false_path -from [get_cells {u_rst_sys/sync_r[0]}]

3.4 单时钟域项目的简化

对于教学项目（单时钟域），不需要 set_clock_groups。但 仍然需要在 SDC 中声明板上所有时钟端口（即使 RTL 中未使用），防止 STA 分析不完整。

4. SDC 约束规范

4.1 最小约束模板（Stage 0/1）

# 主时钟
create_clock -name clk50_in -period 20.000 [get_ports clk50_in]

# 异步输入 false path
set_false_path -from [get_ports rst_n_i]
set_false_path -from [get_ports {btn_n[*]}]

# 慢速输出 false path
set_false_path -to [get_ports led_user]

4.2 完整约束模板

# 1. 所有板上时钟声明
create_clock -name clk50_in     -period 20.000 [get_ports clk50_in]
create_clock -name clk33_ddr    -period 30.000 [get_ports clk33_ddr]

# 2. 时钟不确定性（jitter 预算）
set_clock_uncertainty 0.200 [get_clocks clk50_in]

# 3. 异步时钟组
set_clock_groups -asynchronous \
    -group {clk50_in} \
    -group {clk33_ddr}

# 4. I/O 延迟约束
set_input_delay  -max 5.0 -clock clk50_in [get_ports uart_rxd]
set_output_delay -max 5.0 -clock clk50_in [get_ports uart_txd]

# 5. 异步输入 false path
set_false_path -from [get_ports rst_n_i]
set_false_path -from [get_ports {btn_n[*]}]

# 6. 慢速输出 / 调试输出 false path
set_false_path -to [get_ports led_user]

4.3 约束原则

原则 说明
保守声明 即使 RTL 未使用某些时钟端口，也在 SDC 中声明
异步 = false path 所有异步外部输入加 false path
CDC 首级豁免 CDC 同步链第一级寄存器显式 false path
I/O 按需约束 高速接口用 set_input/output_delay，慢速信号用 false path

4.4 常见错误

错误 后果 修复
SDC 缺少未用时钟端口声明 该端口连接的时钟网络 STA 不分析 始终声明所有板上时钟
未声明时钟组 跨域路径产生虚假 setup 违例 set_clock_groups -asynchronous
同步器首级未 false path metastability 路径被报告为时序违例 精确 false path 到同步器首级
create_clock period 单位错误 20ns 写成 20（ps） 始终用 ns 单位

5. Golden Top 参考约束系统

Golden Top 是 WS-E1 完整硬件参考设计，其约束文件可作为多外设复杂设计的约束模板。

5.1 时钟域声明

# 板上 8 个独立时钟域
create_clock -name clk_50m      -period 20.000 [get_ports clk_50m]
create_clock -name ddr_clk_ref  -period 30.000 [get_ports ddr_clk_ref]
create_clock -name clk_27_mipi_a -period 37.037 [get_ports clk_27_mipi_a]
create_clock -name clk_27_mipi_b -period 37.037 [get_ports clk_27_mipi_b]
create_clock -name pll_ethernet_refclk -period 40.000 [get_ports pll_ethernet_refclk]
create_clock -name rgmii_rxc    -period 8.000  [get_ports rgmii_rxc]

5.2 外设 I/O 约束

# UART (115200 bps)
set_input_delay  -max 10.0 -clock clk_50m [get_ports system_uart_0_io_rxd]
set_output_delay -max 10.0 -clock clk_50m [get_ports system_uart_0_io_txd]

# SPI (10MHz)
set_output_delay -max 20.0 -clock clk_50m [get_ports {system_spi_0_io_*}] -add

# GPIO
set_false_path -from [get_ports {system_gpio_0_io_*}]
set_false_path -to   [get_ports {system_gpio_0_io_*}]

5.3 CDC 同步链豁免

# 复位同步器首级
set_false_path -from [get_cells {u_rst_sys/sync_r[0]}]
set_false_path -from [get_cells {u_rst_spi/sync_r[0]}]

# 跨时钟域信号同步器首级
set_false_path -from [get_cells {u_sync_btn/sync_r[0]}]

6. 仿真规范

6.1 Testbench 模板

`timescale 1ns / 1ps

module tb_module_name;

    reg        clk;
    reg        rst_n;
    wire [7:0] count;

    // DUT 例化
    module_name u_dut (
        .clk_i  (clk),
        .rst_n_i(rst_n),
        .count_o(count)
    );

    // 时钟生成
    always #5 clk = ~clk;  // 100MHz

    // VCD dump
    initial begin
        $dumpfile("dump.vcd");
        $dumpvars(0, tb_module_name);
    end

    // 测试激励
    initial begin
        clk   = 0;
        rst_n = 0;
        #20 rst_n = 1;  // 2 个周期后释放复位
        // ... 激励 ...
        #1000 $finish;
    end

endmodule

6.2 仿真验证要点

  • 复位释放：至少保持 2 个时钟周期
  • 波形 dump：使用 $dumpfile + $dumpvars
  • 测试覆盖：正常操作、边界条件、复位恢复

7. 综合编码注意事项

7.1 避免 Latch

// 错误 — 产生 Latch
always @(*) begin
    if (enable) out = data;
end

// 正确 — 完整 if-else
always @(*) begin
    if (enable) out = data;
    else        out = '0;
end

7.2 避免组合逻辑环

// 错误 — 组合逻辑环
assign a = b | c;
assign b = a & d;

// 正确 — 通过寄存器断开
always @(posedge clk) a_r <= b_r | c;
assign a = a_r;

7.3 寄存器输出优先

// 避免 — 长组合逻辑直接输出到端口
assign led = (complex_combinational_expression);

// 推荐 — 寄存器输出
always @(posedge clk) led_r <= complex_combinational_expression;
assign led = led_r;

7.4 Efinity 综合推断注意事项

  • Efinity efx_map 可以从 RTL 推断 BRAM、DSP、IO 寄存器
  • 使用 (* ram_style = "block" *) 等属性引导推断
  • 统一流程（Unified Flow）中，综合自动推断 GPIO，不要在 ISF 中重复创建综合已推断的 GPIO
  • peri-syn-instantiation / peri-syn-inference 控制外设原语推断行为
高级参考用于按需查询；设计记录、发布流程、审计报告和未来规划不属于客户操作入口。
