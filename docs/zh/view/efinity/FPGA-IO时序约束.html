1. 为什么 IO 时序约束至关重要

核心问题：FPGA 与外部芯片（DDR / PHY / ADC / DAC）通信时，外部信号的到达时间依赖于 PCB trace 长度 + 外部器件 Tco/Tsu/Th——这些信息只有设计者知道，STA 工具不会自动推断。

"Static timing analysis is 'static' because it doesn't depend on specific input stimulus vectors — instead, it analyzes all possible paths in worst case. But there's a key premise: the tool needs to know what 'qualified' means. That standard is what we give via timing constraints." —— Source: [B7] CSDN 时序分析

未约束的后果： - STA 工具假设 0 ns 外部延迟 → 路径过紧 → 违例报告过多 - 或假设无限大延迟 → 路径过松 → 实际硬件不工作

"If you don't add constraints, the tool uses a very permissive default that only guarantees basic connectivity. The tool may route critical signals on slow channels, or use non-optimized placement, ultimately causing the design to fail to work." —— Source: [B7] CSDN 时序分析

两类典型问题：

类型 起点 终点
In-Reg（输入） 外部芯片 → PCB trace → FPGA 输入引脚 FPGA 内部寄存器 D
Reg-Out（输出） FPGA 内部寄存器 Q PCB trace → 外部芯片输入引脚

详见 FPGA-STA基础.md §2。

2. set_input_delay — 输入延迟

作用：告诉 STA 工具外部信号到达 FPGA 输入引脚后还需要多久才能到达内部寄存器 D。

2.1 同步输入（与 FPGA 时钟同源）

# 输入延迟：外部器件 Tco_max + PCB trace max delay
set_input_delay -clock sys_clk -max 4.0 [get_ports adc_data_i*]

# min 延迟：外部器件 Tco_min + PCB trace min delay
set_input_delay -clock sys_clk -min 1.0 [get_ports adc_data_i*]

含义： - -max 4.0（ns）：最坏情况数据延迟 4 ns - STA 工具计算：Required time = sys_clk period − max delay − Tsu − uncertainty - 即：setup slack = (Period − 4 − Tsu − uncertainty) − (Tco + Tlogic + Tnet)

2.2 异步输入（独立时钟域）

# 外部器件有自己的时钟，与 sys_clk 异步
set_input_delay -clock virtual_ext_clk -max 5.0 [get_ports async_in_i]
set_input_delay -clock virtual_ext_clk -min 2.0 [get_ports async_in_i]

# 配合 CDC synchronizer
set_clock_groups -asynchronous -group {sys_clk} -group {virtual_ext_clk}

详见 FPGA-CDC模式.md §3 CDC synchronizer 模板。

2.3 -clock_fall 选项（DDR）

# DDR 接口：下降沿也采样
set_input_delay -clock ddr_clk -clock_fall -max 2.0 [get_ports ddr_dq[*]]

详见 §8。

2.4 -add_delay 多时钟叠加

# 同一输入端口被两个时钟采样（罕见但需注意）
set_input_delay -clock clk_a -max 3.0 [get_ports data_in] -add
set_input_delay -clock clk_b -max 4.0 [get_ports data_in] -add

3. set_output_delay — 输出延迟

作用：告诉 STA 工具 FPGA 输出引脚的数据相对于下游器件时钟边沿的到达时间。

3.1 基本语法

# 最大延迟：外部器件 Tsu
set_output_delay -clock sys_clk -max 2.0 [get_ports led_user_o]

# 最小延迟：外部器件 Th（-min 不能为负）
set_output_delay -clock sys_clk -min -0.5 [get_ports led_user_o]

含义： - -max 2.0（ns）：下游器件需要数据在时钟边沿之后 2 ns 内到达（满足其 Tsu） - STA 工具计算：Required time = sys_clk period + Tco + output max delay - 即：setup slack = (Period + 2 − Tco − output_delay) − (Tco + Tlogic + Tnet + T_output_buf)

3.2 -add_delay 与 -reference_pin

# 指定参考引脚（非时钟端口的 reference）
set_output_delay -reference_pin [get_ports clk_o] -max 1.5 [get_ports data_o*]

4. 源同步接口 Source-Synchronous

定义：时钟与数据同时从外部芯片发出（典型：DDR、RGMII、SPI 高速模式）

4.1 DDR3 接口（AN 042 + DDR3 控制器）

DDR3 控制器是 Efinity 的硬核 IP，约束通常自动生成。手动场景：

# DDR3 时钟：400 MHz（DDR CK 800 MHz data rate）
create_clock -period 2.5 -name ddr_ck [get_ports ddr_ck_t]

# DDR DQ 输入延迟（DDR mode：上升 + 下降沿）
set_input_delay -clock ddr_ck -max 0.4 [get_ports ddr_dq[*]]
set_input_delay -clock ddr_ck -min -0.4 [get_ports ddr_dq[*]]
set_input_delay -clock ddr_ck -clock_fall -max 0.4 [get_ports ddr_dq[*]]
set_input_delay -clock ddr_ck -clock_fall -min -0.4 [get_ports ddr_dq[*]]

# DDR DQ 输出延迟
set_output_delay -clock ddr_ck -max 0.4 [get_ports ddr_dq[*]]
set_output_delay -clock ddr_ck -min -0.4 [get_ports ddr_dq[*]]

Source: AN 042 + DDR3 Soft Controller Core User Guide v1.6 §A4.5

DDR PHY 硬核 (Titanium)：

# Titanium DDR PHY: max 3,300 Mbps
# CLKOUT4 drives controller at 1/4 PHY data rate
create_generated_clock -name ddr_ckout4 \
    -source [get_pins {ddr_phy/clk}] \
    -divide_by 4 \
    [get_pins {ddr_phy/clkout4}]

Source: Titanium DDR DRAM Block User Guide v2.6 §A4.6

4.2 RGMII 接口（AN 043）

RGMII 是源同步 DDR 接口，时钟与数据中心对齐（center-aligned）——时钟边沿在数据眼图中央。

# RGMII TX clock 125 MHz（PLL 输出）
create_generated_clock -name rgmii_tx_clk \
    -source [get_pins {pll_rgmii/clkin}] \
    -multiply_by 5 -divide_by 4 \
    [get_pins {pll_rgmii/clkout}]

# TX data：DDR mode，使用 -clock_fall
set_output_delay -clock rgmii_tx_clk -max 0.5 [get_ports rgmii_txd[*]]
set_output_delay -clock rgmii_tx_clk -min 1.0 [get_ports rgmii_txd[*]]
set_output_delay -clock rgmii_tx_clk -max 0.5 -clock_fall [get_ports rgmii_txd[*]]
set_output_delay -clock rgmii_tx_clk -min 1.0 -clock_fall [get_ports rgmii_txd[*]]

# RX data：从 PHY 输入
create_clock -period 8.0 -name rgmii_rx_clk [get_ports rgmii_rx_clk_i]
set_input_delay -clock rgmii_rx_clk -max 0.5 [get_ports rgmii_rxd[*]]
set_input_delay -clock rgmii_rx_clk -min 1.0 [get_ports rgmii_rxd[*]]
set_input_delay -clock rgmii_rx_clk -clock_fall -max 0.5 [get_ports rgmii_rxd[*]]
set_input_delay -clock rgmii_rx_clk -clock_fall -min 1.0 [get_ports rgmii_rxd[*]]

Source: AN 043 v1.3 §"RGMII Dynamic Clock Switch" ([A4.2])

4.3 SPI 接口（简单源同步）

# SPI 时钟 50 MHz
create_clock -period 20.0 -name spi_clk [get_ports spi_sck_i]

# SPI MOSI/MISO 同步约束
set_input_delay -clock spi_clk -max 5.0 [get_ports spi_miso_i]
set_input_delay -clock spi_clk -min 2.0 [get_ports spi_miso_i]
set_output_delay -clock spi_clk -max 4.0 [get_ports spi_mosi_o]
set_output_delay -clock spi_clk -min 0.0 [get_ports spi_mosi_o]

5. 系统同步接口 System-Synchronous

定义：FPGA 与外部器件共用同一时钟总线（典型：异步 SRAM、并行 DAC、总线接口）

5.1 异步 SRAM 接口

# 系统时钟 100 MHz
create_clock -period 10.0 -name sys_clk [get_ports clk_100m]

# SRAM 输入（地址 + 数据 + 控制）
set_input_delay -clock sys_clk -max 6.0 [get_ports sram_data_io*]
set_input_delay -clock sys_clk -min 1.0 [get_ports sram_data_io*]

# SRAM 输出
set_output_delay -clock sys_clk -max 5.0 [get_ports sram_data_io*]
set_output_delay -clock sys_clk -min 0.0 [get_ports sram_data_io*]
set_output_delay -clock sys_clk -max 5.0 [get_ports {sram_addr[*] sram_oe_n sram_we_n sram_cs_n}]

5.2 总线接口（wishbone / AXI / APB）

# 同源时钟
create_clock -period 10.0 -name sys_clk [get_ports clk_i]

# 输入
set_input_delay -clock sys_clk -max 4.0 [get_ports {bus_data_i[*] bus_ack_i bus_err_i bus_rty_i}]
set_input_delay -clock sys_clk -min 1.0 [get_ports {bus_data_i[*] bus_ack_i bus_err_i bus_rty_i}]

# 输出
set_output_delay -clock sys_clk -max 3.0 [get_ports {bus_data_o[*] bus_adr_o[*] bus_we_o bus_sel_o[*] bus_cyc_o bus_stb_o}]
set_output_delay -clock sys_clk -min 0.0 [get_ports {bus_data_o[*] bus_adr_o[*] bus_we_o bus_sel_o[*] bus_cyc_o bus_stb_o}]

6. FPGA 引脚 Tsu / Th

FPGA 引脚 Tsu/Th：FPGA 内部 IO cell 的固有 setup/hold 要求（与 PCB 无关）。FPGA datasheet 提供。

"The Setup time (t_su) is the time required for the input to be stable before a clock edge." "The Hold time (t_h) is the minimum amount of time required for the input to be stable after the clock edge." —— Source: [B4] MPI Lesson 11

典型 FPGA Tsu/Th：

工艺 Tsu Th
7nm ASIC 45 ps 35 ps
现代 FPGA（典型） 180 ps 120 ps

Source: [B5] arXiv 2510.26985

Efinity Trion/Titanium：参见具体器件 datasheet（TJ180A484S 等）：

Tsu (input setup to global clock) = ~0.6 ns (typical)
Th  (input hold from global clock) = ~0.1 ns (typical)

具体数值由 PnR 工具在 STA 中自动使用。详见对应器件 datasheet。

7. Efinity GPIO Clock Latency

Efinity GPIO 作为时钟输入：需要专门约束时钟 latency。

Source: C:\Efinity\2026.1\doc\topics\tim-clock-latency-gpio.html ([A3])

7.1 时钟 latency 三段

段 起点 终点 来源
板级延迟 外部时钟源 FPGA 引脚 set_clock_latency -source
GCLK 网络延迟 FPGA 引脚（作为 GCLK） 全局时钟树 工具自动推断
PLL 补偿延迟 PLL 输入 PLL 输出 Efinity 自动应用

7.2 完整示例

# 板级时钟延迟：晶振 → FPGA 引脚
set_clock_latency -source -early 1.0 [get_clocks sys_clk]
set_clock_latency -source -late  3.0 [get_clocks sys_clk]

# GCLK 网络延迟（一般工具自动推断，无需手动）
# set_clock_latency -early 0.5 [get_clocks sys_clk]
# set_clock_latency -late  1.0 [get_clocks sys_clk]

详见 FPGA-时钟约束Cookbook.md §5 与 AN 042。

8. SDR vs DDR 时序

8.1 SDR（Single Data Rate）

特点：每个时钟周期 1 bit 数据。

# 上升沿采样：默认
set_input_delay -clock sys_clk -max 4.0 [get_ports sdr_data_i*]
set_input_delay -clock sys_clk -min 1.0 [get_ports sdr_data_i*]

8.2 DDR（Double Data Rate）

特点：每个时钟周期 2 bit 数据（上升沿 + 下降沿采样）。

# DDR 输入：上升 + 下降均采样
set_input_delay -clock ddr_clk -max 1.5 [get_ports ddr_dq[*]]
set_input_delay -clock ddr_clk -min 0.0 [get_ports ddr_dq[*]]
set_input_delay -clock ddr_clk -clock_fall -max 1.5 [get_ports ddr_dq[*]]
set_input_delay -clock ddr_clk -clock_fall -min 0.0 [get_ports ddr_dq[*]]

8.3 中心对齐 vs 边沿对齐

对齐方式 典型接口 set_input_delay 含义
边沿对齐（Edge-Aligned） SPI, async SRAM 数据与时钟同时到达
中心对齐（Center-Aligned） DDR, RGMII 时钟在数据眼图中央

边沿对齐：set_input_delay -max ≈ 数据有效窗口 中心对齐：set_input_delay -max ≈ (周期/2 − setup margin)

8.4 边沿对齐补偿示例（DDR write leveling）

# DDR PHY 自动处理 write leveling；用户无需手动补偿
# 仅在 PHY calibration 失败时手动调整：
set_output_delay -clock ddr_ck -max 0.5 [get_ports ddr_dq[*]]
set_output_delay -clock ddr_ck -min 0.0 [get_ports ddr_dq[*]]

9. 完整 Efinity IO SDC 示例

9.1 综合系统：DDR3 + RGMII + SPI + GPIO

# ============================================
# 时钟定义
# ============================================
create_clock -period 10.0 -name sys_clk [get_ports clk_100m]      # 100 MHz 主时钟
create_generated_clock -name vid_clk -multiply_by 37 -divide_by 25 \
    -source [get_pins {pll_video/clkin}] [get_pins {pll_video/clkout}]   # 148.5 MHz
create_generated_clock -name ddr_ck -multiply_by 4 -divide_by 1 \
    -source [get_pins {pll_ddr/clkin}] [get_pins {pll_ddr/clkout}]       # 400 MHz DDR
create_clock -period 8.0 -name rgmii_rx_clk [get_ports rgmii_rx_clk_i]   # 125 MHz
create_clock -period 20.0 -name spi_sck_i [get_ports spi_sck_i]          # 50 MHz SPI

# ============================================
# 不确定度
# ============================================
set_clock_uncertainty -to sys_clk -setup 0.100
set_clock_uncertainty -to sys_clk -hold  0.020

# ============================================
# 异步分组（视频 + DDR 与主系统隔离）
# ============================================
set_clock_groups -asynchronous \
    -group {sys_clk spi_sck_i rgmii_rx_clk} \
    -group {vid_clk ddr_ck}

# ============================================
# DDR3 控制器（硬核自动约束，部分需手动）
# ============================================
# 硬核 PHY 通常自动生成 set_input_delay
set_input_delay -clock ddr_ck -max 0.4 [get_ports ddr_dq[*]]
set_input_delay -clock ddr_ck -min -0.4 [get_ports ddr_dq[*]]
set_input_delay -clock ddr_ck -clock_fall -max 0.4 [get_ports ddr_dq[*]]
set_input_delay -clock ddr_ck -clock_fall -min -0.4 [get_ports ddr_dq[*]]

# ============================================
# RGMII TX（中心对齐 DDR 输出）
# ============================================
create_generated_clock -name rgmii_tx_clk -multiply_by 5 -divide_by 4 \
    -source [get_pins {pll_rgmii/clkin}] [get_pins {pll_rgmii/clkout}]
set_output_delay -clock rgmii_tx_clk -max 0.5 [get_ports rgmii_txd[*]]
set_output_delay -clock rgmii_tx_clk -min 1.0 [get_ports rgmii_txd[*]]
set_output_delay -clock rgmii_tx_clk -clock_fall -max 0.5 [get_ports rgmii_txd[*]]
set_output_delay -clock rgmii_tx_clk -clock_fall -min 1.0 [get_ports rgmii_txd[*]]

# RGMII RX（中心对齐 DDR 输入）
set_input_delay -clock rgmii_rx_clk -max 0.5 [get_ports rgmii_rxd[*]]
set_input_delay -clock rgmii_rx_clk -min 1.0 [get_ports rgmii_rxd[*]]
set_input_delay -clock rgmii_rx_clk -clock_fall -max 0.5 [get_ports rgmii_rxd[*]]
set_input_delay -clock rgmii_rx_clk -clock_fall -min 1.0 [get_ports rgmii_rxd[*]]

# ============================================
# SPI 输入（外部 SPI Flash）
# ============================================
set_input_delay -clock spi_sck_i -max 5.0 [get_ports spi_miso_i]
set_input_delay -clock spi_sck_i -min 1.0 [get_ports spi_miso_i]
set_output_delay -clock spi_sck_i -max 4.0 [get_ports spi_mosi_o]
set_output_delay -clock spi_sck_i -min 0.0 [get_ports spi_mosi_o]
set_output_delay -clock spi_sck_i -max 4.0 [get_ports spi_csn_o]

# ============================================
# GPIO（按键 + LED）
# ============================================
# 按键输入（异步，用 sys_clk 域同步）
set_input_delay -clock sys_clk -max 8.0 [get_ports btn_i*]
set_input_delay -clock sys_clk -min 2.0 [get_ports btn_i*]

# LED 输出
set_output_delay -clock sys_clk -max 4.0 [get_ports led_o*]
set_output_delay -clock sys_clk -min 0.0 [get_ports led_o*]

# ============================================
# 异步复位 / JTAG / 测试
# ============================================
set_false_path -from [get_ports arst_n]
set_false_path -from [get_ports {tck tms tdi}]
set_false_path -to   [get_ports tdo]

9.2 AN 043 实际 RGMII PHY 模式

# RGMII 2.5 / 25 / 125 MHz 动态切换（Titanium 专用）
create_generated_clock -name rgmii_tx_125 -multiply_by 5 -divide_by 4 \
    -source [get_pins {pll_rgmii/clkin}] [get_pins {pll_rgmii/clkout}]
create_generated_clock -name rgmii_tx_25 -multiply_by 1 -divide_by 4 \
    -source [get_pins {pll_rgmii/clkin}] [get_pins {pll_rgmii/clkout}] -add
create_generated_clock -name rgmii_tx_2p5 -multiply_by 1 -divide_by 40 \
    -source [get_pins {pll_rgmii/clkin}] [get_pins {pll_rgmii/clkout}] -add

# 三选一
set_clock_groups -exclusive \
    -group {rgmii_tx_125} \
    -group {rgmii_tx_25} \
    -group {rgmii_tx_2p5}

Source: AN 043 v1.3 ([A4.2])
